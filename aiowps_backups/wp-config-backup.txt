/**
 * 9Remote / Reverse Proxy Tunnel Temporary Developer Support
 */
if ( getenv( 'WP_REVERSE_PROXY_DEV' ) === 'true' || ( isset( $_SERVER['HTTP_X_FORWARDED_HOST'] ) && strpos( $_SERVER['HTTP_X_FORWARDED_HOST'], 'trycloudflare.com' ) !== false ) ) {
	// 1. Detect Protocol
	$forwarded_proto = $_SERVER['HTTP_X_FORWARDED_PROTO'] ?? 'http';
	$is_ssl          = ( 'https' === strtolower( $forwarded_proto ) );
	if ( $is_ssl ) {
		$_SERVER['HTTPS'] = 'on';
	}

	// 2. Detect Host & Port
	$forwarded_host = $_SERVER['HTTP_X_FORWARDED_HOST'] ?? ( $_SERVER['HTTP_HOST'] ?? 'localhost' );

	// 3. Detect Path Prefix (e.g., /proxy/10028)
	$forwarded_prefix = $_SERVER['HTTP_X_FORWARDED_PREFIX'] ?? '';
	if ( empty( $forwarded_prefix ) && isset( $_SERVER['REQUEST_URI'] ) ) {
		if ( preg_match( '#^(/proxy/\d+)#', $_SERVER['REQUEST_URI'], $matches ) ) {
			$forwarded_prefix = $matches[1];
		}
	}
	$forwarded_prefix = rtrim( $forwarded_prefix, '/' );

	// 4. Set Home and Site URL dynamically
	$scheme     = $is_ssl ? 'https://' : 'http://';
	$mapped_url = $scheme . $forwarded_host . $forwarded_prefix;

	if ( ! defined( 'WP_HOME' ) ) {
		define( 'WP_HOME', $mapped_url );
	}
	if ( ! defined( 'WP_SITEURL' ) ) {
		define( 'WP_SITEURL', $mapped_url );
	}

	// 5. Fix absolute asset URLs pointing to localhost inside content / options
	ob_start( function( $buffer ) use ( $mapped_url ) {
		return str_replace( 'http://localhost:10028', $mapped_url, $buffer );
	} );
}
